class CalculadoraTermodinamica:
    def __init__(self):
        # Constante universal dos gases perfeitos
        self.R_atm = 0.08206  # L·atm/(K·mol)
        self.R_joule = 8.314  # J/(K·mol)

    def gas_ideal(self, P=None, V=None, n=None, T=None):
        """
        Calcula a variável que falta na equação dos gases ideais: PV = nRT.
        P = Pressão (atm)
        V = Volume (L)
        n = Número de mols (mol)
        T = Temperatura (K)
        """
        variaveis_nulas = sum(x is None for x in [P, V, n, T])
        
        if variaveis_nulas != 1:
            raise ValueError("Forneça exatamente três variáveis para calcular a quarta.")

        if P is None:
            return (n * self.R_atm * T) / V
        elif V is None:
            return (n * self.R_atm * T) / P
        elif n is None:
            return (P * V) / (self.R_atm * T)
        elif T is None:
            return (P * V) / (n * self.R_atm)

    def energia_livre_gibbs(self, delta_H, T, delta_S):
        """
        Calcula a variação da Energia Livre de Gibbs: ΔG = ΔH - TΔS
        delta_H = Variação de Entalpia (J ou kJ)
        T = Temperatura (K)
        delta_S = Variação de Entropia (J/K ou kJ/K)
        Retorna ΔG na mesma unidade base de energia fornecida.
        """
        delta_G = delta_H - (T * delta_S)
        
        # Avaliação da espontaneidade
        if delta_G < 0:
            estado = "Espontâneo"
        elif delta_G > 0:
            estado = "Não espontâneo"
        else:
            estado = "Em equilíbrio"
            
        return delta_G, estado

# --- Exemplos de Uso ---
if __name__ == "__main__":
    calc = CalculadoraTermodinamica()

    # 1. Calculando o Volume de 2 mols de um gás a 1 atm e 298 K
    volume = calc.gas_ideal(P=1.0, n=2.0, T=298.15)
    print(f"Volume do gás: {volume:.2f} Litros")

    # 2. Calculando a Energia Livre de Gibbs
    # Exemplo: ΔH = -100000 J (Exotérmica), T = 298 K, ΔS = -150 J/K
    g, espontaneidade = calc.energia_livre_gibbs(delta_H=-100000, T=298.15, delta_S=-150)
    print(f"ΔG: {g:.2f} Joules -> Processo {espontaneidade}")
